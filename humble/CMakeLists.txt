cmake_minimum_required(VERSION 3.16)
project(fast_lio LANGUAGES C CXX)
get_filename_component(FAST_LIO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "${FAST_LIO_ROOT}:msg/Pose6D.msg" DEPENDENCIES geometry_msgs)
include(${FAST_LIO_ROOT}/cmake/fast_lio_core.cmake)
add_executable(fastlio_mapping src/node.cpp)
target_include_directories(fastlio_mapping PRIVATE ${FAST_LIO_ROOT}/include ${FAST_LIO_ROOT}/adapters)
target_link_libraries(fastlio_mapping fast_lio_core)
ament_target_dependencies(fastlio_mapping rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions livox_ros_driver2)
install(TARGETS fastlio_mapping DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz_cfg DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  include(${FAST_LIO_ROOT}/cmake/fast_lio_tests.cmake)
endif()
ament_export_targets(fast_lio_coreTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 PCL)
ament_package()
