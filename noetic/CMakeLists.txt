cmake_minimum_required(VERSION 3.16)
project(fast_lio LANGUAGES C CXX)
get_filename_component(FAST_LIO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions livox_ros_driver message_generation)
add_message_files(DIRECTORY msg BASE_DIR ${FAST_LIO_ROOT}/msg FILES Pose6D.msg)
generate_messages(DEPENDENCIES geometry_msgs)
# Preserve the original ROS 1 matching worker default.
set(FAST_LIO_DEFAULT_MATCH_THREADS 1)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86)|(X86)|(amd64)|(AMD64)")
  include(ProcessorCount)
  ProcessorCount(FAST_LIO_CPU_COUNT)
  if(FAST_LIO_CPU_COUNT GREATER 4)
    set(FAST_LIO_DEFAULT_MATCH_THREADS 3)
  elseif(FAST_LIO_CPU_COUNT GREATER 3)
    set(FAST_LIO_DEFAULT_MATCH_THREADS 2)
  endif()
endif()
include(${FAST_LIO_ROOT}/cmake/fast_lio_core.cmake)
catkin_package(LIBRARIES fast_lio_core CATKIN_DEPENDS message_runtime)
add_executable(fastlio_mapping src/node.cpp)
target_include_directories(fastlio_mapping PRIVATE ${FAST_LIO_ROOT}/include ${FAST_LIO_ROOT}/adapters ${catkin_INCLUDE_DIRS})
target_link_libraries(fastlio_mapping fast_lio_core ${catkin_LIBRARIES})
add_dependencies(fastlio_mapping ${catkin_EXPORTED_TARGETS} ${${PROJECT_NAME}_EXPORTED_TARGETS})
install(TARGETS fastlio_mapping RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config rviz_cfg DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  include(${FAST_LIO_ROOT}/cmake/fast_lio_tests.cmake)
endif()
