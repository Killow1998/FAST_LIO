cmake_minimum_required(VERSION 3.16)
project(fast_lio_core LANGUAGES C CXX)
get_filename_component(FAST_LIO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)

  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(${FAST_LIO_ROOT}/cmake/fast_lio_core.cmake)

if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  include(${FAST_LIO_ROOT}/cmake/fast_lio_tests.cmake)
endif()
